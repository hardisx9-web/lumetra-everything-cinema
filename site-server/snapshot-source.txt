function active(p='s'){return `${p}.checked_ms <= ? AND ${p}.expires_ms > ?`;}
const now=()=>Date.now();
function selection(t){return `w.id,w.title,w.original_title,w.alternate_titles,w.year,w.director,w.countries,w.runtime,w.verification_level,w.coverage_tier,EXISTS(SELECT 1 FROM source s WHERE s.work_id=w.id AND s.watch_state='WATCH_HERE' AND s.checked_ms<=${t} AND s.expires_ms>${t}) embedded,EXISTS(SELECT 1 FROM source s WHERE s.work_id=w.id AND s.checked_ms<=${t} AND s.expires_ms>${t}) watchable`;}
function decode(r){return {...r,countries:JSON.parse(r.countries||'[]'),alternate_titles:JSON.parse(r.alternate_titles||'[]'),watchable:!!r.watchable,embedded:!!r.embedded};}
export async function querySnapshot(pool,params,validator){const f=validator(params),t=now(),args=[],terms=[],q=f.q,country=params.get('country'),year=params.get('year');if(q){const escaped=q.replace(/[\\%_]/g,c=>'\\'+c);const phrase=q.split(/\s+/).filter(Boolean).map(s=>'"'+s.replaceAll('"','""')+'"*').join(' AND ');terms.push(`(w.id IN (SELECT id FROM search WHERE search MATCH ?) OR w.title LIKE ? ESCAPE '\\' OR w.original_title LIKE ? ESCAPE '\\' OR w.alternate_titles LIKE ? ESCAPE '\\' OR w.director LIKE ? ESCAPE '\\')`);args.push(phrase,...Array(4).fill('%'+escaped+'%'));}if(country){terms.push('EXISTS(SELECT 1 FROM json_each(w.countries) WHERE value=?)');args.push(country);}if(year){terms.push('w.year=?');args.push(Number(year));}if(params.get('watch')==='1')terms.push(`EXISTS(SELECT 1 FROM source s WHERE s.work_id=w.id AND s.checked_ms<=${t} AND s.expires_ms>${t})`);const where=terms.length?' WHERE '+terms.join(' AND '):'';const total=pool.prepare('SELECT count(*) n FROM work w'+where).get(...args).n;const items=pool.prepare(`SELECT ${selection(t)} FROM work w${where} ORDER BY w.title,w.id LIMIT ? OFFSET ?`).all(...args,f.limit,(f.page-1)*f.limit).map(decode);return {items,total,page:f.page,limit:f.limit,snapshot:true};}
export async function snapshotDossier(pool,id){if(!/^[\w-]{1,160}$/.test(id))throw Object.assign(new Error('Invalid film ID'),{status:400});const t=now();const r=pool.prepare(`SELECT ${selection(t)} FROM work w WHERE w.id=?`).get(id);if(!r)throw Object.assign(new Error('Film not found'),{status:404});const work=decode(r);const sources=pool.prepare(`SELECT payload FROM source s WHERE work_id=? AND ${active()}`).all(id,t,t).map(r=>JSON.parse(r.payload));const evidence=pool.prepare('SELECT DISTINCT payload FROM evidence WHERE work_id=? LIMIT 30').all(id).map(r=>JSON.parse(r.payload));const related=pool.prepare(`SELECT ${selection(t)} FROM work w WHERE w.id<>? AND ((? IS NOT NULL AND w.director=?) OR (? IS NOT NULL AND w.year=?)) ORDER BY CASE WHEN w.director=? THEN 0 ELSE 1 END,w.title LIMIT 8`).all(id,work.director,work.director,work.year,work.year,work.director).map(decode);return {...work,sources,evidence,related};}
export async function snapshotStats(pool){const t=now(),base=JSON.parse(pool.prepare("SELECT value FROM meta WHERE key='counts'").get().value);return {...base,...pool.prepare(`SELECT count(*) eligible_sources,count(DISTINCT work_id) watchable_works,count(CASE WHEN watch_state='WATCH_HERE' THEN 1 END) embedded_sources FROM source s WHERE ${active()}`).get(t,t),as_of:new Date(t).toISOString(),snapshot_exported_at:pool.prepare("SELECT value FROM meta WHERE key='exported_at'").get().value};}
export async function snapshotFacets(pool){return {countries:pool.prepare('SELECT DISTINCT j.value FROM work w,json_each(w.countries) j ORDER BY j.value').all().map(r=>r.value),years:pool.prepare('SELECT DISTINCT year value FROM work WHERE year IS NOT NULL ORDER BY year DESC').all().map(r=>r.value)};}
